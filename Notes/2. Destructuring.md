#### 1. Array Destructuring
##### Basic example
```
const arr = [10, 20, 30];

const [a, b, c] = arr;

console.log(a, b, c); // 10 20 30
```
##### Skip values
```
const [a, , c] = [10, 20, 30];

console.log(a, c); // 10 30
```
##### Default values
```
const [a = 5, b = 10] = [1];

console.log(a, b); // 1 10
```
##### Swap variables
```
let x = 1;
let y = 2;

[x, y] = [y, x];

console.log(x, y); // 2 1
```
#### 2. Object Destructuring

##### Basic example
```
const user = {
  name: "Saravanan",
  age: 25
};

const { name, age } = user;

console.log(name, age);
```
##### Rename variables
```
const { name: userName } = user;

console.log(userName);
```
##### Default values
```
const { city = "Chennai" } = user;

console.log(city);
```

##### Nested destructuring
```
const user = {
  profile: {
    name: "Saravanan"
  }
};

const { profile: { name } } = user;

console.log(name);
```

#### 3. Function Parameters
```
function greet({ name, age }) {
  console.log(`Hello ${name}, age ${age}`);
}

greet({ name: "Saravanan", age: 25 });
```
#### 4. Real Use Case
```
const user = {
  name: "Saravanan",
  skills: ["JS", "React"]
};

const { name, skills: [firstSkill] } = user;

console.log(firstSkill); // JS
```
#### 5. With Rest Operator
```
const [a, ...rest] = [1, 2, 3, 4];

console.log(rest); // [2, 3, 4]
```
#### Summary
- Extract values easily
- Works with arrays [] and objects {}
- Supports default values & renaming
- Useful in functions and APIs
- Cleaner, modern syntax

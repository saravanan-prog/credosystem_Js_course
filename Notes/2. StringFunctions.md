#### 1. length → count letters

``` 
    var string = "Hello World";
    var length =  string.length;
    console.log("string Length is =====>",length)                  👉 Output : 11

```

#### 2. toUpperCase() → CAPITAL

```
 var string = "Hello World";
 var upperCase = string.toUpperCase()
 console.log("Transformed Uppercase======>", upperCase)                👉 Output : HELLO WORLD"

```

#### 3. toLowerCase() → small letters

```
 var string = "Hello World";
 var lowerCase = string.toLowerCase()  
 console.log("Transformed Lowercase ======>",lowerCase)                  👉 Output : "hello world"

```
#### 4. includes() → check word

```
var str = "Hello World";
var findStr = str.includes("Hello")   
console.log(findStr)                     👉 Output : true
```





#### 5. indexOf() → position

```
let str = "Hello World";
let findIndex = str.indexOf("World")  
console.log(findIndex)                     👉 Output : 6
```

#### 5.1 lastIndexof() → last position

```
let str = "Hello World World";
let findIndex = str.lastIndexof("World")  
console.log(findIndex)                     👉 Output : 12
```


#### 6. slice(startpos,endpos + 1 ) → cut string


```
let str = "Hello World";
let slicedStr = str.slice(0, 5)     
console.log(slicedString)                     👉 Output : "Hello"
```



#### 7. replace() → change text


```
let str = "Hello World";
let newStr = str.replace("World", "JS")  
console.log(newStr)                           👉 Output : "Hello JS"
```



#### 8. trim() → remove spaces


```
let str = " hi ";
let trimmedStr = str.trim();  
console.log(trimmedStr)      👉 Output : "hi"

```

#### 9. split() → convert to array

```
let str = "Hello World";
let splitStr = str.split(" ");  
console.log(splitStr)           👉 Output : ["Hello", "World"]

```


#### 10. charAt() → get character

```
let str = "Hello World";
let char = str.charAt(1);  
console.log(char)           👉 Output : "e"

```

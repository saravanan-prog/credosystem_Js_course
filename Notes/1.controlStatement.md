# 1. if, else, else if


✅ Basic if

```
    let age = 20;

    if (age >= 18) {
        console.log("Adult");
    }

```


✅ if...else

```
let age = 16;

if (age >= 18) {
    console.log("Adult");
} else {
    console.log("Minor");
}

```


✅ else if

```
    let marks = 75;

    if (marks >= 90) {
        console.log("A");
    } else if (marks >= 70) {
        console.log("B");
    } else {
        console.log("C");
    }


```


🟡 2. switch

👉 Used when checking multiple fixed values

```
let day = 2;

switch (day) {
    case 1:
        console.log("Monday");
        break;
    case 2:
        console.log("Tuesday");
        break;
    default:
        console.log("Invalid");
}

```
